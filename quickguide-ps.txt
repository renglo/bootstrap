# Bootstrap quickguide — PowerShell
# Prerequisites assumed: AWS CLI, Python 3.12, Node+CDK, Git, WSL
# Paste into an empty folder (creates infra-installer/ and runs from there).

# --- Vars (edit these) ---
$ENV = "<env>"
$AWS_PROFILE = "<aws-profile>"
$AWS_REGION = "us-east-1"   # deploy-time only (CLI / CDK); not in customer-config
$REPO = "<Org/releases-repo>"
$EXTENSION_REPO = "https://github.com/<Org>/<extension-repo>"
$EXTENSION_PATH = "<extension-repo>"

# --- 1. Clone repos ---
New-Item -ItemType Directory -Force -Path infra-installer | Out-Null
Set-Location infra-installer

git clone https://github.com/renglo/bootstrap
git clone https://github.com/renglo/launcher
git clone https://github.com/renglo/extensions-service
git clone $EXTENSION_REPO

# Workspace root — keep all commands here. $py absolute for synth only.
$Root = (Get-Location).Path
$py = Join-Path $Root "bootstrap\venv\Scripts\python.exe"
$CdkOut = "bootstrap\output\$ENV\cdk"   # cloud assembly from install.py synth

# --- 2. Virtualenvs ---
wsl bash bootstrap/setup-venvs.sh

# --- 3. customer-config.json ---
# Account/region are NOT in this file — templates use AWS::AccountId / AWS::Region.
# Choose the target with $AWS_PROFILE / $AWS_REGION at deploy time.
$cfg = @"
{
    "env_name": "$ENV",
    "github_repo": "$REPO",
    "extension_path": "$EXTENSION_PATH",
    "enable_staging": true,
    "compute_type": "ec2",
    "ec2_instance_type": "t3.xlarge",
    "ec2_min_instances": 1,
    "ec2_desired_instances": 1,
    "ec2_max_instances": 1
}
"@
Set-Content -Path (Join-Path $Root "launcher\cdk\customer-config.json") -Value $cfg -Encoding utf8

# --- 4. Synth (from workspace root) ---
& $py (Join-Path $Root "bootstrap\install.py") synth

# --- 5. Deploy stack-a (from workspace root) ---
# --app points at the pre-synthesized cloud assembly (manifest.json), so CDK
# does not re-run Python and you do not need to cd into $CdkOut.
# Stack id is just "$ENV-stack-a" — not a filesystem path.
cdk deploy "$ENV-stack-a" `
  --app $CdkOut `
  --require-approval never `
  --profile $AWS_PROFILE `
  --region $AWS_REGION

# If new account (no token.actions.githubusercontent.com OIDC), add:
#   --parameters "CreateGitHubOIDC=true"

# Note: stack-a builds and pushes the seed image automatically (CodeBuild
# custom resource). No manual step is needed before stack-b.
# To rebuild the seed image manually later:
#   aws codebuild start-build --project-name "$ENV-seed-image" `
#     --profile $AWS_PROFILE --region $AWS_REGION

# --- 6. Deploy stack-b (from workspace root) ---
# Default HandlersNetworkMode=create (dedicated VPC). For an existing VPC, see §6b.
cdk deploy "$ENV-stack-b" `
  --app $CdkOut `
  --exclusively `
  --require-approval never `
  --profile $AWS_PROFILE `
  --region $AWS_REGION

# --- 6b. stack-b with existing VPC (compute_type=ec2) ---
